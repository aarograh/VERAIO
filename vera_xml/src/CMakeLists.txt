INCLUDE(TribitsLibraryMacros)

SET(HEADERS "")
SET(SOURCES "")

ASSERT_DEFINED(TeuchosCore_ENABLE_MPI)
IF (TeuchosCore_ENABLE_MPI)
  SET(CTEUCHOS_HAVE_MPI ON)
ENDIF()

ASSERT_DEFINED(CTeuchos_ENABLE_TeuchosComm)
IF (CTeuchos_ENABLE_TeuchosComm)
  SET(CTEUCHOS_HAVE_TEUCHOSCOMM ON)
ENDIF()

SET_AND_INC_DIRS(DIR ${CMAKE_CURRENT_SOURCE_DIR})
APPEND_GLOB(HEADERS ${DIR}/*.hpp)
APPEND_GLOB(SOURCES ${DIR}/*.cpp)

INCLUDE_DIRECTORIES(${CMAKE_CURRENT_BINARY_DIR})

TRIBITS_ADD_LIBRARY(
  vera_xml
  HEADERS ${HEADERS}
  SOURCES ${SOURCES}
  )

SET_TARGET_PROPERTIES(vera_xml PROPERTIES LINKER_LANGUAGE C)
