
INCLUDE(TribitsPackageMacros)

TRIBITS_PACKAGE(VERAIn)

ADD_SUBDIRECTORY(src)

TRIBITS_ADD_TEST_DIRECTORIES(test)

# Install the reactor-aware ASCII input parser

INSTALL(
  PROGRAMS scripts/react2xml.pl
  DESTINATION "${${PROJECT_NAME}_INSTALL_RUNTIME_DIR}"
  COMPONENT ${PACKAGE_NAME}
  )

INSTALL(
  DIRECTORY scripts/Packages scripts/Templates
  DESTINATION "${${PROJECT_NAME}_INSTALL_RUNTIME_DIR}"
  COMPONENT ${PACKAGE_NAME}
  )  

# Install some examples

INSTALL(
  DIRECTORY install/example/
  DESTINATION "share/VERAIn/example"
  COMPONENT ${PACKAGE_NAME}
  PATTERN "*.sh" PERMISSIONS OWNER_EXECUTE OWNER_WRITE OWNER_READ GROUP_EXECUTE GROUP_READ
  )

# Install documentation

INSTALL(
  DIRECTORY docs/
  DESTINATION "share/VERAIn/doc"
  COMPONENT ${PACKAGE_NAME}
  )

# Install documentation

INSTALL(
  FILES install/README.VERAIn
  DESTINATION "."
  COMPONENT ${PACKAGE_NAME}
  )

TRIBITS_PACKAGE_POSTPROCESS()
